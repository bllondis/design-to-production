export const tier = (name, ...path) =>
  tokens.filter((token) => token.tier === name && path.every((part, index) => token.path[index] === part));
export const swatches = (items) => Object.fromEntries(items.map((token) => [token.path.at(-1), token.value]));
export const cssVar = (path) => `--${path.replaceAll(".", "-")}`;

export const TokenTable = ({ items }) => (
  <table>
    <thead>
      <tr>
        <th>Token</th>
        <th>Value</th>
        <th>Points at</th>
        <th>Use</th>
      </tr>
    </thead>
    <tbody>
      {items.map((token) => (
        <tr key={token.name}>
          <td>
            <code>{`--${token.name}`}</code>
          </td>
          <td>
            {token.type === "color" && (
              <span
                aria-hidden="true"
                style={{
                  display: "inline-block",
                  width: "0.9rem",
                  height: "0.9rem",
                  marginRight: "0.5rem",
                  verticalAlign: "middle",
                  border: "1px solid rgba(128, 128, 128, 0.5)",
                  background: token.value,
                }}
              />
            )}
            {String(token.value)}
          </td>
          <td>{token.reference ? <code>{cssVar(token.reference)}</code> : "—"}</td>
          <td>{token.description ?? ""}</td>
        </tr>
      ))}
    </tbody>
  </table>
);

# Tokens

Every colour, size and motion value comes from `tokens/*.tokens.json` (W3C DTCG format).
`npm run build -w @tetiana/tokens` builds them with Style Dictionary into `@tetiana/tokens/tokens.css`.
They come in three tiers, mirrored one-to-one by the Figma variable collections Primitives, Semantic and Component.
Each tier points at the one before it, and components use only semantic and component tokens, never primitives.

<TierTable />

## Primitives

The raw palette, families, weights and timings. Components and pages never use them directly.

<ColorPalette>
  {["ink", "paper", "blue", "amber", "alpha"].map((palette) => (
    <ColorItem
      key={palette}
      title={palette}
      subtitle={`--color-${palette}-*`}
      colors={swatches(tier("primitive", "color", palette))}
    />
  ))}
</ColorPalette>

## Semantic

Roles that pages and components use. Most point at a primitive; sizes that only make sense as a role hold their own value.

## Component

Decisions that belong to one component, such as a button's hover colour. Each points at a semantic token, so a change
of role reaches the component without editing it.
