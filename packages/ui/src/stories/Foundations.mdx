import { Meta, ColorPalette, ColorItem, Typeset } from "@storybook/addon-docs/blocks";
import tokens from "@tetiana/tokens/tokens.json";

<Meta title="Foundations/Tokens" />

export const group = (...path) => tokens.filter((token) => path.every((part, index) => token.path[index] === part));
export const swatches = (items) => Object.fromEntries(items.map((token) => [token.path.at(-1), token.value]));

# Tokens

Every colour, size and motion value comes from `tokens/*.tokens.json` (W3C DTCG format).
`npm run build -w @tetiana/tokens` builds them with Style Dictionary into `@tetiana/tokens/tokens.css`; there are three tiers: primitives (raw values), semantic (what a value is for) and component (the decisions one component makes). Components use semantic and component tokens, never primitives.

## Colour roles

<ColorPalette>
  {["surface", "text", "border", "accent", "light"].map((role) => (
    <ColorItem key={role} title={role} subtitle={`--color-${role}-*`} colors={swatches(group("color", role))} />
  ))}
</ColorPalette>
